An array of Devanagari numerals indexed `0` through `9`, used for formatting numbers into Nepali script.

---

## Import

```typescript
import { NUMBER_NP } from "nepali-date-library";
```

---

## Value

```typescript
export const NUMBER_NP = ["०", "१", "२", "३", "४", "५", "६", "७", "८", "९"];
```

## Examples

### Converting Arabic Digits to Devanagari

```typescript
import { NUMBER_NP } from "nepali-date-library";

function toNepaliDigits(num: number | string): string {
  return String(num)
    .split("")
    .map((ch) => {
      const code = ch.charCodeAt(0) - 48;
      return code >= 0 && code <= 9 ? NUMBER_NP[code] : ch;
    })
    .join("");
}

console.log(toNepaliDigits(2081)); // "२०८१"
console.log(toNepaliDigits("2081-01-25")); // "२०८१-०१-२५"
```
